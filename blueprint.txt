        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        function playSFX(type) {
            if (!audioCtx) return;
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);

            if (type === 'beep') {
                osc.frequency.setValueAtTime(800, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.1);
            } else if (type === 'splash') {
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(150, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.3);
                gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.3);
            }
        }

                state.fishes.forEach(f => {
                    if (f.isHeld) return;

                        minDist = d;
                        nearest = f;
                    }
                });

                    logPhysics(`${p.label} grabbed ${nearest.name}`);

        function updatePhysics() {
            const speed = 0.8;

            // Player movements
            state.players.forEach(p => {
                if (p.keys.up) p.vy -= speed;
                if (p.keys.down) p.vy += speed;
                if (p.keys.left) p.vx -= speed;
                if (p.keys.right) p.vx += speed;

                p.vx *= state.friction;
                p.vy *= state.friction;

                p.x += p.vx;
                p.y += p.vy;

                // Deck boundary
                p.x = Math.max(120, Math.min(680, p.x));
                p.y = Math.max(100, Math.min(380, p.y));

                if (p.holding) {
                    p.holding.x = p.x;
                    p.holding.y = p.y - 20;
                }
            });

            // Calculate boat center of mass & tilt
            let portMass = 0;
            let starboardMass = 0;

            const allObjects = [...state.players, ...state.fishes];
            allObjects.forEach(obj => {
                if (obj.x < 400) portMass += 1;
                else starboardMass += 1;
            });

            const massDiff = (starboardMass - portMass);
            state.boatAngle = massDiff * 1.5 * state.tiltSensitivity;

            document.getElementById('boat-angle-indicator').textContent = 
                `Tilt: ${state.boatAngle.toFixed(1)}° (${state.boatAngle > 0 ? 'Starboard' : state.boatAngle < 0 ? 'Port' : 'Stable'})`;

            // Apply gravity slide based on tilt angle
            const slideForce = (state.boatAngle / 10);

            state.fishes.forEach(f => {
                if (!f.isHeld) {
                    f.vx += slideForce;
                    f.vx *= state.friction;
                    f.vy *= state.friction;

                    f.x += f.vx;
                    f.y += f.vy;

                    // Cooler collision check
                    const cb = state.coolerBox;
                    if (f.x > cb.x && f.x < cb.x + cb.w && f.y > cb.y && f.y < cb.y + cb.h) {
                        state.coolerCount++;
                        document.getElementById('cooler-count').textContent = state.coolerCount;
                        logPhysics(`SECURED! ${f.name} placed in Cooler.`);
                        playSFX('beep');
                        f.x = -999; // Remove
                    }

                    // Overboard check
                    if (f.x < 100 || f.x > 700 || f.y < 80 || f.y > 400) {
                        if (f.x !== -999) {
                            state.overboardCount++;
                            document.getElementById('overboard-count').textContent = state.overboardCount;
                            logPhysics(`SPLASH! ${f.name} fell overboard!`);
                            playSFX('splash');
                            f.x = -999;
                        }
                    }
                }
            });

            state.fishes = state.fishes.filter(f => f.x !== -999);
        }

        function renderCanvas() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Water background
            ctx.fillStyle = '#071728';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Save transform for boat tilt
            ctx.save();
            ctx.translate(canvas.width / 2, canvas.height / 2);
            ctx.rotate((state.boatAngle * Math.PI) / 180);
            ctx.translate(-canvas.width / 2, -canvas.height / 2);

            // Draw Boat Hull
            ctx.fillStyle = '#1e293b';
            ctx.strokeStyle = '#334155';
            ctx.lineWidth = 6;
            ctx.beginPath();
            ctx.roundRect(100, 80, 600, 320, 40);
            ctx.fill();
            ctx.stroke();

            // Deck Wood Planks Lines
            ctx.strokeStyle = '#0f172a';
            ctx.lineWidth = 2;
            for (let y = 110; y < 390; y += 30) {
                ctx.beginPath();
                ctx.moveTo(120, y);
                ctx.lineTo(680, y);
                ctx.stroke();
            }

            // Draw Cooler Box Station
            const cb = state.coolerBox;
            ctx.fillStyle = '#0284c7';
            ctx.strokeStyle = '#38bdf8';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.roundRect(cb.x, cb.y, cb.w, cb.h, 12);
            ctx.fill();
            ctx.stroke();

            ctx.fillStyle = '#ffffff';
            ctx.font = 'bold 12px Plus Jakarta Sans';
            ctx.textAlign = 'center';
            ctx.fillText('COOLER 🧊', cb.x + cb.w / 2, cb.y + cb.h / 2 + 4);

            // Render Fishes
            state.fishes.forEach(f => {
                if (!f.isHeld) {
                    ctx.font = '24px sans-serif';
                    ctx.textAlign = 'center';
                    ctx.fillText(f.emoji, f.x, f.y);
                }
            });

            // Render Players
            state.players.forEach(p => {
                // Shadow
                ctx.fillStyle = 'rgba(0,0,0,0.3)';
                ctx.beginPath();
                ctx.ellipse(p.x, p.y + 12, 16, 8, 0, 0, Math.PI * 2);
                ctx.fill();

                // Body
                ctx.fillStyle = p.color;
                ctx.beginPath();
                ctx.arc(p.x, p.y, 16, 0, Math.PI * 2);
                ctx.fill();

                // Label
                ctx.fillStyle = '#0f172a';
                ctx.font = 'bold 10px Plus Jakarta Sans';
                ctx.textAlign = 'center';
                ctx.fillText(p.label, p.x, p.y + 3);

                // Held item offset render
                if (p.holding) {
                    ctx.font = '20px sans-serif';
                    ctx.fillText(p.holding.emoji, p.x, p.y - 22);
                }
            });

            ctx.restore();
        }
